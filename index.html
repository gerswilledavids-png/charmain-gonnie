/** * GONNIE FLEET MASTER - Capture handler for the charmain-gonnie web app * * Save this as its own file (Capture). It does not touch your other scripts. * Writes ONLY the input columns on the 'Trip Log' tab and leaves every * formula column (Trip ID, Fleet No, Distance, KM/L, profit, etc.) untouched. * Row 1 = title, row 2 = headers, data starts on row 3. */ var GCAP_TRIP_SHEET = "Trip Log"; var GCAP_FIRST_ROW = 3; var GCAP_LAST_ROW = 999; // formulas are pre-filled down to this row /* ---------- POST: log a new trip ---------- */ function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(20000); var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(GCAP_TRIP_SHEET); if (!sheet) throw new Error("Tab '" + GCAP_TRIP_SHEET + "' not found"); var row = gcapNextFreeRow_(sheet); if (row > GCAP_LAST_ROW) throw new Error("Trip Log is full (row " + GCAP_LAST_ROW + ")"); // Start destination: use it directly, or take the part before the arrow in "A -> B" var start = data.start_destination || data.start || ""; if (!start && data.route) { start = String(data.route).split(/\s*(?:→|->)\s*/)[0]; } // Trip Notes feeds the Customer column (text after the first colon) var notes = String(data.notes || ""); if (data.customer) notes = (notes ? notes + " | " : "") + "Customer: " + data.customer; // D..K Registration, Driver, Phone, Odo Start, Odo End, Fuel (L), Cost/L, No.Trips sheet.getRange(row, 4, 1, 8).setValues([[ String(data.registration || "").toUpperCase().trim(), String(data.driver_name || "").toUpperCase().trim(), data.driver_phone || "", gcapNum_(data.odo_start), gcapNum_(data.odo_end), gcapNum_(data.fuel_used_l), gcapNum_(data.cost_per_liter), gcapNum_(data.no_trips) || 1 ]]); // R..T Trip Notes, Manager, Manager Phone sheet.getRange(row, 18, 1, 3).setValues([[ notes, data.manager || "", data.manager_phone || "" ]]); // W..X Start Destination, Revenue (Y = End Destination is a formula) sheet.getRange(row, 23, 1, 2).setValues([[start, gcapNum_(data.revenue)]]); // Z..AB Cargo Type, Load KG, Fuel Station (AC GPS Verified left blank) sheet.getRange(row, 26, 1, 3).setValues([[ data.cargo_type || "", gcapNum_(data.load_kg), data.fuel_station || "" ]]); // AN Paid Status sheet.getRange(row, 40).setValue(data.paid_status || "Pending"); // AT Fuel Filled on Return (L), AW Fill Type sheet.getRange(row, 46).setValue(gcapNum_(data.fuel_filled_return)); sheet.getRange(row, 49).setValue(data.fill_type || "Metered"); // B Date last - it switches the row's formulas on (Trip ID = ROW()-2) sheet.getRange(row, 2).setValue(gcapParseDate_(data.date)); SpreadsheetApp.flush(); return gcapJson_({ status: "success", trip_id: row - 2, row: row, message: "Trip logged to Gonnie Fleet Master" }); } catch (err) { return gcapJson_({ status: "error", message: err.toString() }); } finally { try { lock.releaseLock(); } catch (x) {} } } /* ---------- GET: ?action=ping or ?action=recent ---------- */ function doGet(e) { try { var action = (e && e.parameter) ? e.parameter.action : ""; if (action === "ping") { return gcapJson_({ status: "success", message: "Gonnie capture endpoint is live" }); } if (action === "recent") { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(GCAP_TRIP_SHEET); if (!sheet) throw new Error("Tab '" + GCAP_TRIP_SHEET + "' not found"); var rows = sheet.getRange(GCAP_FIRST_ROW, 1, GCAP_LAST_ROW - GCAP_FIRST_ROW + 1, 11).getValues(); var trips = []; for (var i = 0; i < rows.length; i++) { var r = rows[i]; if (r[1] === "") continue; // skip empty rows trips.push({ trip_id: r[0], date: r[1], registration: r[3], driver_name: r[4], odo_start: r[6], odo_end: r[7] }); } return gcapJson_({ status: "success", trips: trips.slice(-50) }); } return gcapJson_({ status: "error", message: "Invalid or missing action parameter." }); } catch (err) { return gcapJson_({ status: "error", message: err.toString() }); } } /* ---------- Helpers ---------- */ function gcapNextFreeRow_(sheet) { var n = GCAP_LAST_ROW - GCAP_FIRST_ROW + 1; var dates = sheet.getRange(GCAP_FIRST_ROW, 2, n, 1).getValues(); var last = -1; for (var i = 0; i < n; i++) if (dates[i][0] !== "") last = i; return GCAP_FIRST_ROW + last + 1; // row after the last trip } function gcapNum_(v) { if (v === undefined || v === null || v === "") return ""; var n = Number(String(v).replace(/\s/g, "").replace(",", ".")); return isNaN(n) ? "" : n; } function gcapParseDate_(v) { if (v instanceof Date) return v; if (typeof v === "string" && /^\d{4}-\d{2}-\d{2}$/.test(v)) { var p = v.split("-"); return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2])); } var d = v ? new Date(v) : new Date(); if (isNaN(d.getTime())) d = new Date(); return new Date(d.getFullYear(), d.getMonth(), d.getDate()); } function gcapJson_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
